cmake_minimum_required(VERSION 3.22)
project(aril_efo LANGUAGES C CXX ASM)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "Static linkage" FORCE)
add_subdirectory(deps/dobby EXCLUDE_FROM_ALL)
add_library(profile SHARED native/profile.cpp)
target_include_directories(profile PRIVATE deps/zygisk/module/jni)
target_compile_features(profile PRIVATE cxx_std_17)
target_compile_options(profile PRIVATE -Wall -Wextra -fvisibility=hidden)
target_link_options(profile PRIVATE -Wl,-z,max-page-size=16384 -Wl,--exclude-libs,ALL)
target_link_libraries(profile PRIVATE dobby log dl)
